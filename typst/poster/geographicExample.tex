\begin{tikzpicture}[scale=2]
\tikzstyle{vertex}=[circle,inner sep=0mm,minimum size=3ex]
\def\n{5}
\def\regionOneSize{3}
\def\radius{3cm}
%draw k-complete graph
\foreach \i in {1,...,\n} {
    \def\vertexcolour{unsw@green}
    \ifnum \i > \regionOneSize
    \def\vertexcolour{unsw@red}
    \fi
    \pgfmathsetmacro{\angle}{162 - (\i - 1) * (180/ \n)}
    \node[vertex, fill=\vertexcolour] (v\i) at (\angle:\radius) {};
}
% draw edges, latencies of us-east-1, us-east-2, us-west-2, ap-northeast-1, ap-southeast-1
\foreach \i/\j/\latency in {%
        1/2/14ms, 1/3/69ms, 1/4/149ms, 1/5/221ms,
        2/3/57ms, 2/4/136ms, 2/5/209ms,
        3/4/106ms, 3/5/181ms, 4/5/70ms} {%
    \def\edgecolour{unsw@blue}
    \ifnum \i > \regionOneSize \else
        \ifnum \j > \regionOneSize \def\edgecolour{unsw@orange}\fi
    \fi
    \draw[line width=5pt, color=\edgecolour] (v\i) -- (v\j);
    \draw (v\i) -- (v\j) node[midway, above, sloped, font=\scriptsize] {\latency};
}
\end{tikzpicture}
